    int A = 1;
    int B = 1;

    double W11 = 1.0;
    double W12 = 1.0;
    double b1 = 0.5;

    double W21 = 1.0;
    double W22 = 1.0;
    double b2 = -1.5;

    double W31 = 1.0;
    double W32 = 1.0;
    double b3 = -1.0;

    double h1 = sigmoid((A * W11) + (B * W12) + b1);
    double h2 = sigmoid((A * W21) + (B * W22) + b2);

    double output = sigmoid((h1 * W31) + (h2 * W32) + b3);

    int finalOutput = (output <= 0.5) ? 0 : 1;

    std::cout << finalOutput;
